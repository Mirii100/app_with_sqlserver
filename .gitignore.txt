  Future<void> _handleContinue() async {
    debugPrint("DEBUG: _handleContinue called");
    if (_isLoading) {
      debugPrint("DEBUG: _isLoading is true, returning");
      return;
    }

    final hasInternet = await ConnectivityUtil.ensureOnline();
    if (!hasInternet) {
      debugPrint("DEBUG: No internet");
      Navigator.of(context).push(
        MaterialPageRoute(builder: (context) => const NoInternetScreen()),
      );
      return;
    }

    final step = _steps[_currentStep];

    for (final field in step.fields) {
      final label = field['label']!;
      final controller = _controllers[label];
      final value = controller?.text.trim() ?? '';
      if (value.isEmpty) {
        debugPrint("DEBUG: Validation failed for $label");
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Please enter $label')),
        );
        return;
      }
      // ... (rest of validation)
    }

    debugPrint("DEBUG: Validation passed");
    if (step.hasUploads && (_idPhoto == null || _selfie == null)) {
      debugPrint("DEBUG: Missing uploads");
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Please upload ID photo and take a selfie')),
      );
      return;
    }

    if (_currentStep < _steps.length - 1) {
      debugPrint("DEBUG: Moving to next step");
      setState(() => _currentStep++);
    } else {
      debugPrint("DEBUG: Calling _createAccount");
      await _createAccount();
    }
  }

  Future<void> _createAccount() async {
    debugPrint("DEBUG: _createAccount called");
    setState(() => _isLoading = true);

    final hasInternet = await ConnectivityUtil.ensureOnline();
    if (!hasInternet) {
      if (mounted) {
        setState(() => _isLoading = false);
        Navigator.of(context).push(
          MaterialPageRoute(builder: (context) => const NoInternetScreen()),
        );
      }
      return;
    }

    final email = _emailController.text.trim();
    final password = _passwordController.text;
    final fullName = _fullNameController.text.trim();
    final phoneNumber = _phoneController.text.trim();
    final nationalId = _nationalIdController.text.trim();
    final county = _countyController.text.trim();
    final town = _townController.text.trim();
    final postalCode = _postalCodeController.text.trim();
    final employmentType = _employmentTypeController.text.trim();
    final monthlyIncome = _monthlyIncomeController.text.trim();

    try {
      // NOTE: signUpWithEmail's profilePhoto/idPhoto/selfiePhoto params must
      // now be typed XFile? in BackendAuthService (they were File? before).
      final user = await _authService.signUpWithEmail(
        email: email,
        password: password,
        fullName: fullName,
        phoneNumber: phoneNumber,
        profilePhoto: _selfie,
        idPhoto: _idPhoto,
        selfiePhoto: _selfie,
      );

      debugPrint("DEBUG: account created, user=$user");

      if (user != null && mounted) {
        // From this point on the account already exists in the backend
        // (the signup call above returned 201). Any failure in the
        // follow-up calls below should NOT block the success message or
        // redirect — otherwise the user sees an error, thinks nothing
        // happened, and taps "Create account" again, which then fails
        // with a duplicate-email error since the account already exists.
        await _safeCall(() => _authService.saveUserData(
          userId: user.uid,
          fullName: fullName,
          phoneNumber: phoneNumber,
          nationalId: nationalId,
          email: email,
          county: county,
          town: town,
          postalCode: postalCode,
          employmentType: employmentType,
          monthlyIncome: monthlyIncome,
          balance: 0.0,
        ), 'saveUserData');

        await _safeCall(() => _authService.createSavingsGoal(
          userId: user.uid,
          goalId: 'new_laptop',
          title: 'New laptop',
          purpose: 'Education & upskilling',
          targetAmount: 67000,
          savedAmount: 0,
          autoSaveAmount: 0,
          autoSaveEnabled: false,
        ), 'createSavingsGoal');

        final now = DateTime.now();
        final monthKey = '${now.year}-${now.month.toString().padLeft(2, '0')}';
        await _safeCall(() => _authService.saveBudgetData(
          userId: user.uid,
          monthKey: monthKey,
          budgetData: {
            'categories': {
              'Food': 14200,
              'Transport': 8600,
              'Airtime': 5100,
              'Other': 9800,
            },
            'budgetLimits': {
              'Food': 20000,
              'Transport': 10000,
              'Airtime': 4500,
              'Other': 15000,
            },
            'month': monthKey,
          },
        ), 'saveBudgetData');

        await _safeCall(() => _authService.saveLoanProducts(user.uid, [
          {
            'id': 'quick_cash',
            'name': 'Quick Cash',
            'description': 'Up to KSh 35,000 · 30 days · 4.5% flat',
            'isBestMatch': true,
            'isOutline': false,
          },
          {
            'id': 'business_boost',
            'name': 'Business Boost',
            'description': 'Up to KSh 250,000 · 12 months · 2.1% p.m.',
            'isBestMatch': false,
            'isOutline': true,
          },
          {
            'id': 'asset_finance',
            'name': 'Asset Finance',
            'description': 'Up to KSh 1.2M · 36 months · Collateralised',
            'isBestMatch': false,
            'isOutline': true,
          },
        ]), 'saveLoanProducts');

        await _safeCall(() => _authService.saveUserLoanLimit(
          userId: user.uid,
          limit: 35000,
          used: 0,
        ), 'saveUserLoanLimit');

        await _safeCall(() => _authService.saveBillers(user.uid, [
          {'id': 'kplc_prepaid', 'icon': 'bell', 'title': 'KPLC Prepaid', 'subtitle': 'Meter •• 88213'},
          {'id': 'riverbank_academy', 'icon': 'book', 'title': 'Riverbank Academy', 'subtitle': 'Term 3 fees'},
          {'id': 'ncba_bank', 'icon': 'bank', 'title': 'NCBA Bank', 'subtitle': 'Merchant payments'},
          {'id': 'safaricom', 'icon': 'card', 'title': 'Safaricom', 'subtitle': 'Airtime & data'},
          {'id': 'nairobi_water', 'icon': 'home', 'title': 'Nairobi Water', 'subtitle': 'Utility bill'},
          {'id': 'gojane_kg', 'icon': 'repeat', 'title': 'GoJane KG', 'subtitle': 'Subscription'},
        ]), 'saveBillers');

        await _safeCall(() => _authService.saveSecuritySettings(
          userId: user.uid,
          pin: null,
          biometricEnabled: false,
        ), 'saveSecuritySettings');

        // Show success message, then redirect to Login screen.
        // This runs regardless of whether any of the follow-up calls
        // above succeeded, because the account itself is already created.
        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text('Account created successfully! Please login.'),
              backgroundColor: Colors.green,
              duration: Duration(seconds: 2),
            ),
          );

          await Future.delayed(const Duration(seconds: 2));

          if (!mounted) return;

          Navigator.of(context).pushAndRemoveUntil(
            MaterialPageRoute(
              builder: (context) => const LoginScreen(),
            ),
            (route) => false,
          );
        }
      } else {
        // signUpWithEmail returned null unexpectedly.
        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text('Account creation failed. Please try again.'),
              backgroundColor: Colors.red,
            ),
          );
        }
      }
    } on BackendApiException catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(e.message),
            backgroundColor: Colors.red,
          ),
        );
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text('Error: $e'),
            backgroundColor: Colors.red,
          ),
        );
      }
    } finally {
      if (mounted) {
        setState(() => _isLoading = false);
      }
    }
  }

  /// Runs a follow-up backend call and swallows any failure so it can't
  /// block the account-creation success flow. The account already exists
  /// by the time these run; a failure here just means some non-critical
  /// setup data (goals, budget defaults, etc.) didn't save and can be
  /// retried later from within the app.
  Future<void> _safeCall(Future<void> Function() call, String label) async {
    try {
      await call();
    } catch (e) {
      debugPrint('DEBUG: $label failed (non-fatal): $e');
    }
  }

  Widget _buildStepper() {
    final isDark = Theme.of(context).brightness == Brightness.dark;
    return Row(
      children: List.generate(_steps.length, (index) {
        final isActive = index == _currentStep;
        return Expanded(
          child: Row(
            children: [
              Container(
                width: 22,
                height: 22,
                decoration: BoxDecoration(
                  color: isActive
                      ? (isDark ? AppColors.darkPrimary2 : AppColors.primary2)
                      : (isDark ? AppColors.darkBorder : AppColors.border),
                  shape: BoxShape.circle,
                ),
                child: Center(
                  child: Text(
                    '${index + 1}',
                    style: TextStyle(
                      fontSize: 11,
                      fontWeight: FontWeight.w700,
                      color: isActive
                          ? Colors.white
                          : (isDark ? AppColors.darkMuted : AppColors.muted),
                    ),
                  ),
                ),
              ),
              if (index < _steps.length - 1)
                Expanded(
                  child: Container(
                    height: 2,
                    color: isDark ? AppColors.darkBorder : AppColors.border,
                  ),
                ),
            ],
          ),
        );
      }),
    );
  }

  Widget _buildUploadBox({
    required AlexiaIconType icon,
    required String title,
    required String subtitle,

  }) {
    return GestureDetector(

          border: Border.all(
            color: isDark ? AppColors.darkBorder : AppColors.border,
            width: 1,
          ),
          borderRadius: BorderRadius.circular(14),
        ),
        child: Row(
          children: [
            Container(
              width: 40,
              height: 40,
              decoration: BoxDecoration(
                color: (isDark ? AppColors.darkPrimary2 : AppColors.primary2)
                    .withOpacity(0.1),
                shape: BoxShape.circle,
              ),
              child: Center(
                child: AlexiaIcon(
                  iconType: icon,
                  size: 20,
                  color: isDark ? AppColors.darkPrimary2 : AppColors.primary2,
                ),
              ),
            ),
            const SizedBox(width: 12),
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    title,
                    style: TextStyle(
                      fontSize: 12.5,
                      fontWeight: FontWeight.w600,
                      color: isDark ? AppColors.darkInk : AppColors.ink,
                    ),
                  ),
                  Text(
                    subtitle,
                    style: TextStyle(
                      fontSize: 11,
                      color: isDark ? AppColors.darkMuted : AppColors.muted,
                    ),
                  ),
                ],
              ),
            ),
          ],
        ),
      ),
    );
  }
}
